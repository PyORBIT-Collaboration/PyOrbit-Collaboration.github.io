   int EllipseApertureShape::inside(Bunch* bunch, int count){
   
       if(x_half_axis == 0. || y_half_axis == 0.){
           return 0;
       }
   
       double** coord = bunch->coordArr();
   
       double x = (coord[count][0] - x_center)/x_half_axis;
       double y = (coord[count][2] - y_center)/y_half_axis;
       double r2 = x*x + y*y;
       if(r2 <= 1.0){
           return 1;
       }
       return 0;
   }
